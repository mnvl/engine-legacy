
CMAKE_MINIMUM_REQUIRED (VERSION 2.6)
PROJECT (ENGINE)

INCLUDE_DIRECTORIES (${ENGINE_SOURCE_DIR})

IF (WIN32)
	FOREACH (DIRECTORY "c:\\c++libs" "d:\\c++libs" "e:\\c++libs")
		IF (EXISTS ${DIRECTORY})
			SET (CPP_LIBRARIES_DIRECTORY ${DIRECTORY})
		ENDIF()
	ENDFOREACH()
	
	IF (NOT CPP_LIBRARIES_DIRECTORY)
		MESSAGE (FATAL_ERROR "can't find c++libs directory")
	ENDIF()

	MESSAGE ("-- using " ${CPP_LIBRARIES_DIRECTORY})

	INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\boost\\include")
	LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\boost\\lib")
	
	INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\cppunit\\include")
	LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\cppunit\\lib")
	
	INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\xerces-c\\src")
	LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\xerces-c\\lib")

	INCLUDE_DIRECTORIES ("$ENV{DXSDK_DIR}\\include")
	LINK_DIRECTORIES ("$ENV{DXSDK_DIR}\\lib\\x86")

	INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\lua\\src")
	LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}\\lua\\lib")
ENDIF()

ADD_SUBDIRECTORY (bootstrap)
