
CMAKE_MINIMUM_REQUIRED(VERSION 2.6)
PROJECT(ENGINE)

SET(ENGINE_CMAKE_DIR "${ENGINE_SOURCE_DIR}/etc/cmake")
SET(ENGINE_LUA_DIR "${ENGINE_SOURCE_DIR}/lua")

LIST(APPEND CMAKE_MODULE_PATH "${ENGINE_CMAKE_DIR}")

SET(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
SET(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)

ENABLE_TESTING()

SET(ENGINE_LIBRARIES)

INCLUDE(CollectSourceFiles)
INCLUDE(bin2c)

IF(WIN32)
  ADD_DEFINITIONS(-D_CRT_SECURE_NO_WARNINGS)
  ADD_DEFINITIONS(-D_WIN32_WINNT=0x0501)
  ADD_DEFINITIONS(-DD3D_DEBUG_INFO)

  FOREACH(DIRECTORY "c:/c++libs" "d:/c++libs" "e:/c++libs")
    IF(EXISTS ${DIRECTORY})
      SET(CPP_LIBRARIES_DIRECTORY ${DIRECTORY})
    ENDIF()
  ENDFOREACH()

  IF(NOT CPP_LIBRARIES_DIRECTORY)
    MESSAGE(FATAL_ERROR "can't find c++libs directory")
  ENDIF()

  MESSAGE(STATUS "using " ${CPP_LIBRARIES_DIRECTORY})

  FILE(GLOB BOOST_INCLUDEDIR "${CPP_LIBRARIES_DIRECTORY}/boost/include/boost-*/")
  SET(BOOST_LIBRARYDIR "${CPP_LIBRARIES_DIRECTORY}/boost/lib")
  SET(Boost_USE_STATIC_LIBS ON)

  SET(PROTOBUF_ROOT "${CPP_LIBRARIES_DIRECTORY}/protobuf")

  SET(ENV{LIBPQXX_ROOT} "${CPP_LIBRARIES_DIRECTORY}/libpqxx")

  FILE(GLOB XERCESC_ROOT "${CPP_LIBRARIES_DIRECTORY}/xerces-c-*/")
  MESSAGE(STATUS "XERCESC_ROOT = " ${XERCESC_ROOT})
  INCLUDE_DIRECTORIES("${XERCESC_ROOT}/include")
  LINK_DIRECTORIES("${XERCESC_ROOT}/lib")
  LIST(APPEND ENGINE_LIBRARIES debug xerces-c_3D optimized xerces-c_3)

  ADD_DEFINITIONS(-DDIRECTINPUT_VERSION=0x0800)
  INCLUDE_DIRECTORIES("$ENV{DXSDK_DIR}/include")
  LINK_DIRECTORIES("$ENV{DXSDK_DIR}/lib/x86")
ENDIF()

IF(UNIX)
  ADD_DEFINITIONS(-DBOOST_ALL_DYN_LINK)

  FIND_LIBRARY(ENGINE_LIBM m)
  LIST(APPEND ENGINE_LIBRARIES ${ENGINE_LIBM})
ENDIF()

INCLUDE_DIRECTORIES(${ENGINE_SOURCE_DIR}/src)
INCLUDE_DIRECTORIES(${ENGINE_SOURCE_DIR}/build/src)

SET(Boost_USE_MULTITHREADED ON)
FIND_PACKAGE(Boost 1.40 REQUIRED COMPONENTS system thread date_time regex filesystem unit_test_framework)
INCLUDE_DIRECTORIES(${Boost_INCLUDE_DIRS})
LIST(APPEND ENGINE_LIBRARIES ${Boost_LIBRARIES})

FIND_PACKAGE(ProtocolBuffers)

IF(PROTOBUF_FOUND)
  INCLUDE_DIRECTORIES(${PROTOBUF_INCLUDE_DIR})
  LIST(APPEND ENGINE_LIBRARIES ${PROTOBUF_LIBRARY})
  ADD_DEFINITIONS(-DPROTOBUF_FOUND)
ENDIF()

FIND_PACKAGE(Maya COMPONENTS Foundation OpenMaya)

INCLUDE_DIRECTORIES(contrib/lua/src contrib/luabind)
LIST(APPEND ENGINE_LIBRARIES lua luabind LuaBitOp)
ADD_SUBDIRECTORY(contrib)

ADD_SUBDIRECTORY(src)
