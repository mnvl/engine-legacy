
CMAKE_MINIMUM_REQUIRED (VERSION 2.6)
PROJECT (ENGINE)

SET (ENGINE_CMAKE_DIR "${ENGINE_SOURCE_DIR}/etc/cmake")
LIST (APPEND CMAKE_MODULE_PATH "${ENGINE_CMAKE_DIR}")
SET (EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
SET (LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)

SET (CMAKE_BUILD_TYPE Debug)

INCLUDE (AddTests)

IF (WIN32)
  ADD_DEFINITIONS (-D_CRT_SECURE_NO_WARNINGS)

  FOREACH (DIRECTORY "c:/c++libs" "d:/c++libs" "e:/c++libs")
    IF (EXISTS ${DIRECTORY})
      SET (CPP_LIBRARIES_DIRECTORY ${DIRECTORY})
    ENDIF()
  ENDFOREACH()

  IF (NOT CPP_LIBRARIES_DIRECTORY)
    MESSAGE (FATAL_ERROR "can't find c++libs directory")
  ENDIF()

  MESSAGE ("-- using " ${CPP_LIBRARIES_DIRECTORY})

  SET (BOOST_ROOT "${CPP_LIBRARIES_DIRECTORY}/boost")
  SET (BOOST_LIBRARYDIR "${CPP_LIBRARIES_DIRECTORY}/boost/lib")
  SET (Boost_USE_STATIC_LIBS ON)
  
  SET (PROTOBUF_ROOT "${CPP_LIBRARIES_DIRECTORY}/protobuf")

  INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/cppunit/include")
  LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/cppunit/lib")

  ADD_DEFINITIONS (-DXERCES_STATIC_LIBRARY)
  INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/xerces-c/src")
  LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/xerces-c/lib")

  ADD_DEFINITIONS (-DDIRECTINPUT_VERSION=0x0800)
  INCLUDE_DIRECTORIES ("$ENV{DXSDK_DIR}/include")
  LINK_DIRECTORIES ("$ENV{DXSDK_DIR}/lib/x86")

  INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/lua/src")
  LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/lua/lib/static")

  INCLUDE_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/luabind")
  LINK_DIRECTORIES ("${CPP_LIBRARIES_DIRECTORY}/luabind/lib")
ENDIF()

INCLUDE_DIRECTORIES (${ENGINE_SOURCE_DIR})

ENABLE_TESTING()

FIND_PACKAGE(ProtocolBuffers)
INCLUDE_DIRECTORIES (${PROTOBUF_INCLUDE_DIR})

FIND_PACKAGE(Boost 1.35 REQUIRED COMPONENTS system thread)
INCLUDE_DIRECTORIES (${Boost_INCLUDE_DIRS})

IF (WIN32)
  ADD_SUBDIRECTORY (graphic)
  ADD_SUBDIRECTORY (platform)
ENDIF()

ADD_SUBDIRECTORY (math)
ADD_SUBDIRECTORY (server)
ADD_SUBDIRECTORY (util)

ADD_SUBDIRECTORY (starmasters)
