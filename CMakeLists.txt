
CMAKE_MINIMUM_REQUIRED(VERSION 2.6)
PROJECT(ENGINE)

SET(ENGINE_ROOT ${ENGINE_SOURCE_DIR})
ADD_DEFINITIONS(-DENGINE_ROOT="${ENGINE_ROOT}")

SET(ENGINE_CMAKE_DIR "${ENGINE_ROOT}/etc/cmake")
SET(ENGINE_LUA_DIR "${ENGINE_ROOT}/lua")

LIST(APPEND CMAKE_MODULE_PATH "${ENGINE_CMAKE_DIR}")

SET(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
SET(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)

ENABLE_TESTING()

SET(ENGINE_LIBRARIES)

INCLUDE(CollectSourceFiles)
INCLUDE(AddConfigurationDependentTest)
INCLUDE(bin2c)

IF(WIN32)
  ADD_DEFINITIONS(-D_CRT_SECURE_NO_WARNINGS)
  ADD_DEFINITIONS(-D_WIN32_WINNT=0x0501)
  ADD_DEFINITIONS(-DD3D_DEBUG_INFO)

  IF(NOT ("$ENV{TOOLS_DIR}" STREQUAL ""))
  	SET(TOOLS_DIR $ENV{TOOLS_DIR})
    MESSAGE(STATUS "TOOLS_DIR = " ${TOOLS_DIR})
  ELSE()	 
    SET(TOOLS_DIR "d:/tools")
  ENDIF()

  FILE(GLOB BOOST_ROOT "${TOOLS_DIR}/boost_*/")
  SET(BOOST_LIBRARYDIR "${BOOST_ROOT}/stage/lib")
  SET(Boost_USE_STATIC_LIBS ON)
  MESSAGE(STATUS "BOOST_ROOT = " ${BOOST_ROOT})

  FILE(GLOB XERCESC3_ROOT "${TOOLS_DIR}/xerces-c-3.*/")
  MESSAGE(STATUS "XERCESC3_ROOT = " ${XERCESC3_ROOT})

  MESSAGE(STATUS "ENV{DXSDK_DIR} = " $ENV{DXSDK_DIR})
  ADD_DEFINITIONS(-DDIRECTINPUT_VERSION=0x0800)
  INCLUDE_DIRECTORIES("$ENV{DXSDK_DIR}/include")
  LINK_DIRECTORIES("$ENV{DXSDK_DIR}/lib/x86")
ENDIF()

IF(UNIX)
  ADD_DEFINITIONS(-DBOOST_ALL_DYN_LINK)

  FIND_LIBRARY(ENGINE_LIBM m)
  LIST(APPEND ENGINE_LIBRARIES ${ENGINE_LIBM})
ENDIF()

INCLUDE_DIRECTORIES(${ENGINE_ROOT}/src)
INCLUDE_DIRECTORIES(${ENGINE_ROOT}/build/src)

SET(Boost_USE_MULTITHREADED ON)
FIND_PACKAGE(Boost 1.42 REQUIRED COMPONENTS system thread date_time regex filesystem unit_test_framework)
INCLUDE_DIRECTORIES(${Boost_INCLUDE_DIRS})
LIST(APPEND ENGINE_LIBRARIES ${Boost_LIBRARIES})

FIND_PACKAGE(XercesC3 REQUIRED)
INCLUDE_DIRECTORIES(${XERCESC3_INCLUDE_DIR})
LIST(APPEND ENGINE_LIBRARIES ${XERCESC3_LIBRARIES})

FIND_PACKAGE(Maya COMPONENTS Foundation OpenMaya)

ADD_DEFINITIONS(-DLOG4CPLUS_STATIC)
INCLUDE_DIRECTORIES(contrib/lua/src contrib/luabind contrib/log4cplus/include)
ADD_SUBDIRECTORY(contrib)
LIST(APPEND ENGINE_LIBRARIES lua luabind LuaBitOp log4cplus)

ADD_SUBDIRECTORY(src)
ADD_SUBDIRECTORY(lua)
